def clamp(x,a=0,b=1): return max(a,min(b,x))
def ease(x): x=clamp(x);return x*x*(3-2*x)
def event(t,at,duration=0.8):return ease((t-at)/duration)
def tween_keys(t,keys):
    if t<=keys[0][0]:return keys[0][1]
    for (a,x),(b,y) in zip(keys,keys[1:]):
        if t<=b:return x+(y-x)*ease((t-a)/(b-a))
    return keys[-1][1]
def px(x):return round(x*S)
def rgb(col):
    if isinstance(col,tuple):return col
    return tuple(int(col[i:i+2],16) for i in (1,3,5))

def text(im,x,y,value,size=14,color=WHITE,bold=False,anchor='lt',light=False):

def width(value,size=14,bold=False):return font(size,bold).getlength(str(value))/S
def wrapped(im,x,y,value,max_width,size=14,color=WHITE,line_height=None,max_lines=3,bold=False):
    line_height=line_height or size*1.45
    words=value.split(); lines=[];line=''
    for word in words:
        candidate=(line+' '+word).strip()
        if width(candidate,size,bold)>max_width and line:lines.append(line);line=word
        else:line=candidate
    if line:lines.append(line)
    for idx,line in enumerate(lines[:max_lines]):text(im,x,y+idx*line_height,line,size,color,bold)
    return min(len(lines),max_lines)*line_height

@lru_cache(maxsize=128)
def asset(name):return Image.open(ASSETS/name).convert('RGB')
@lru_cache(maxsize=100)
def cover(name,w,h,position=(.5,.5)):
    return ImageOps.fit(asset(name),(int(w),int(h)),method=Image.Resampling.LANCZOS,centering=position)
def picture(im,name,box,r=0,position=(.5,.5)):
    x,y,x2,y2=[px(v) for v in box];w=x2-x;h=y2-y
    if w<=0 or h<=0:return
    bit=cover(name,w,h,position)
    if r:
        mask=Image.new('L',(w,h));ImageDraw.Draw(mask).rounded_rectangle((0,0,w,h),radius=px(r),fill=255)
        im.paste(bit,(x,y),mask)
    else:im.paste(bit,(x,y))

@lru_cache(maxsize=48)
def shade(w,h,axis,stops):
    arr=np.arange(w if axis=='x' else h,dtype=np.float32)
    vals=np.interp(arr,np.array([a for a,b in stops])*(w if axis=='x' else h),[b for a,b in stops])
    alpha=np.broadcast_to(vals[None,:] if axis=='x' else vals[:,None],(h,w)).astype(np.uint8)
    layer=np.zeros((h,w,4),dtype=np.uint8);layer[:,:,3]=alpha
    return Image.fromarray(layer,'RGBA')
def scrim(im,box,axis='x',stops=((0,235),(.35,210),(.75,0),(1,0)),r=0):
    x,y,x2,y2=[px(v) for v in box];w=x2-x;h=y2-y
    layer=shade(w,h,axis,tuple(stops))
    if r:
        mask=Image.new('L',(w,h));ImageDraw.Draw(mask).rounded_rectangle((0,0,w,h),radius=px(r),fill=255)
        a=np.minimum(np.asarray(mask),np.asarray(layer.getchannel('A'))).astype('uint8')
        layer=layer.copy();layer.putalpha(Image.fromarray(a))
    im.paste(layer,(x,y),layer)
def base(color='#090B10'):return Image.new('RGB',(W,H),rgb(color))

def svg_shapes(path):
    root=ET.parse(path).getroot();vb=list(map(float,root.attrib['viewBox'].split()));shapes=[]
    for e in root.iter('{http://www.w3.org/2000/svg}path'):
        tokens=re.findall(r'[a-zA-Z]|[-+]?(?:\d*\.\d+|\d+\.?\d*)(?:[eE][-+]?\d+)?',e.attrib['d'])
        i=0;cmd='';cur=(0.,0.);start=cur;points=[];sub=[]
        while i<len(tokens):
            if tokens[i].isalpha():cmd=tokens[i];i+=1
            rel=cmd.islower();c=cmd.upper()
            if c=='Z':
                if points:sub.append(points);points=[]
                cur=start;cmd='';continue
            n={'M':2,'L':2,'H':1,'V':1,'C':6}.get(c)
            if n is None:raise ValueError('Unsupported SVG path command '+cmd)
            v=list(map(float,tokens[i:i+n]));i+=n
            if c in ('M','L'):
                dest=(v[0]+(cur[0] if rel else 0),v[1]+(cur[1] if rel else 0))
                if c=='M':
                    if points:sub.append(points)
                    points=[];start=dest;cmd='l' if rel else 'L'
                points.append(dest);cur=dest
            elif c=='H':cur=(v[0]+(cur[0] if rel else 0),cur[1]);points.append(cur)
            elif c=='V':cur=(cur[0],v[0]+(cur[1] if rel else 0));points.append(cur)
            elif c=='C':
                p0=cur;p1=(v[0]+(cur[0] if rel else 0),v[1]+(cur[1] if rel else 0));p2=(v[2]+(cur[0] if rel else 0),v[3]+(cur[1] if rel else 0));p3=(v[4]+(cur[0] if rel else 0),v[5]+(cur[1] if rel else 0))
                for u in np.linspace(0,1,20)[1:]:points.append(tuple((1-u)**3*p0[k]+3*(1-u)**2*u*p1[k]+3*(1-u)*u*u*p2[k]+u**3*p3[k] for k in range(2)))
                cur=p3
        if points:sub.append(points)
        col=e.attrib.get('fill',RED);col='#B1060F' if col.startswith('url') else col
        shapes.append((sub,col))
    return vb,shapes

def build_audio():
    sr=48000;n=sr*LENGTH;music=np.zeros((n,2),np.float32);rng=np.random.default_rng(7228);beat=60/90;bar=beat*4
    chords=[(45,[57,60,64,67]),(41,[53,57,60,64]),(48,[55,60,64,67]),(43,[55,59,62,69])]
    def hz(note):return 440*2**((note-69)/12)
    def add(start,signal,pan=0):
        a=max(0,int(start*sr));b=min(n,a+len(signal));p=signal[:b-a]
        if len(p):music[a:b,0]+=p*math.sqrt((1-pan)/2);music[a:b,1]+=p*math.sqrt((1+pan)/2)
    for bi in range(math.ceil(LENGTH/(bar*2))):
        start=bi*bar*2;dur=min(bar*2+1,LENGTH-start);tt=np.arange(int(dur*sr))/sr
        bass,notes=chords[bi%4];env=np.minimum(1,tt/1.0)*np.minimum(1,(dur-tt)/1.1)
        for k,note in enumerate(notes):
            f=hz(note);pad=(np.sin(2*np.pi*f*tt)+.35*np.sin(2*np.pi*f*1.003*tt)+.13*np.sin(4*np.pi*f*tt))*env*.032
            add(start,pad,(-.42,.38,-.18,.25)[k])
        add(start,np.sin(2*np.pi*hz(bass)*tt)*env*.044)
    # Gentle pulse and evolving arpeggio, all synthesized specifically for this film.
    for bi in range(int(LENGTH/beat)):
        start=bi*beat;section=int(start/(bar*2));bass,notes=chords[section%4];tt=np.arange(int(.75*sr))/sr
        energy=.52+.24*math.sin(math.pi*start/LENGTH)
        kick=np.sin(2*np.pi*(43*tt+4*(1-np.exp(-tt*32))))*np.exp(-tt*13)*.05*energy;add(start,kick)
        note=notes[(bi//2)%4]+12;tt=np.arange(int(1.2*sr))/sr
        pluck=(np.sin(2*np.pi*hz(note)*tt)+.18*np.sin(2*np.pi*hz(note)*2*tt))*np.exp(-tt*4.8)*np.minimum(1,tt/.015)*.022*energy
        if bi%2==0:add(start,pluck,.48*math.sin(bi*.4));add(start+.25,pluck*.30,-.48*math.sin(bi*.4))
    # Original soft D-pad tick and select chime at staged interactions.
    ticks=[13,17,20,31.8,36.8,39.5,46.8,49.5,51.8,55,58.8,68.8,71.8,74.8,77.8,86.8,90,97,108,111,114,120,121.6,123.2,124.8,129,141,153,154,157,161,167,169,172]
    for at in ticks:
        tt=np.arange(int(.085*sr))/sr;sound=(np.sin(2*np.pi*890*tt)+.24*np.sin(2*np.pi*1335*tt))*np.exp(-tt*62)*np.minimum(1,tt/.006)*.09;add(at,sound)
    for scene in SCENES[1:]:
        tt=np.arange(int(.7*sr))/sr;noise=rng.normal(0,1,len(tt)).astype(np.float32);noise=np.convolve(noise,np.ones(15)/15,'same');env=np.sin(np.pi*tt/.7)**2
        add(scene[0]-.20,noise*env*.05,-.15)
    fade=np.minimum(1,np.arange(n)/(sr*2))*np.minimum(1,(n-1-np.arange(n))/(sr*3));music*=fade[:,None]
    peak=float(np.max(np.abs(music)));music*=min(1,.72/max(.001,peak))
    with wave.open(str(ROOT/'original-soundtrack.wav'),'wb') as w:w.setnchannels(2);w.setsampwidth(2);w.setframerate(sr);w.writeframes((music*32767).astype('<i2').tobytes())
    print(f'Original stereo soundtrack: {LENGTH}s, peak {peak:.3f}',flush=True)

def render_video(preview=False):

    dur=18 if preview else LENGTH
    cmd=[str(FFMPEG),'-y','-hide_banner','-loglevel','warning','-f','rawvideo','-vcodec','rawvideo','-pix_fmt','rgb24','-s',f'{W}x{H}','-r',str(FPS),'-i','pipe:0','-i',str(ROOT/'original-soundtrack.wav'),'-i',str(ROOT/'chapters.ffmetadata'),'-map','0:v:0','-map','1:a:0','-map_metadata','2','-map_chapters','2','-c:v','libx264','-preset','fast','-crf','19','-threads','4','-pix_fmt','yuv420p','-c:a','aac','-b:a','192k','-ar','48000','-movflags','+faststart','-t',str(dur),str(temp)]
    log=(ROOT/'render.log').open('w',encoding='utf-8');proc=subprocess.Popen(cmd,stdin=subprocess.PIPE,stdout=subprocess.DEVNULL,stderr=log,creationflags=0x08000000|0x00004000)
    started=time.monotonic();last=started;total=FPS*dur
    try:
        for i in range(total):
            t=(i/FPS if not preview else (i/FPS)*10)
            proc.stdin.write(frame(t).tobytes())
            now=time.monotonic()
            if now-last>=15 or i==total-1:
                print(f'Rendered {i+1}/{total} frames · {100*(i+1)/total:.1f}% · elapsed {now-started:.0f}s',flush=True);last=now
        proc.stdin.close();code=proc.wait();log.close()
        if code:raise RuntimeError('FFmpeg failed; see render.log')
        temp.replace(target);print('SAVED '+str(target),flush=True)
    except BaseException:
        if proc.poll() is None:proc.terminate()
        log.close();raise

def render_segment(index):
    """Independent frames can be rendered concurrently without holding a film in RAM."""
    if os.name=='nt':
        import ctypes
        ctypes.windll.kernel32.SetPriorityClass(ctypes.windll.kernel32.GetCurrentProcess(),0x4000)
    cache=ROOT/'.render-cache';cache.mkdir(exist_ok=True)
    dest=cache/f'{index:02d}-{SCENES[index][2]}.mp4'
    if dest.exists() and dest.stat().st_size>10000:return str(dest)
    temp=dest.with_name(dest.stem+'.rendering.mp4')
    start,end=SCENES[index][:2];total=(end-start)*FPS
    cmd=[str(FFMPEG),'-y','-hide_banner','-loglevel','error','-f','rawvideo','-vcodec','rawvideo','-pix_fmt','rgb24','-s',f'{W}x{H}','-r',str(FPS),'-i','pipe:0','-an','-c:v','libx264','-preset','veryfast','-crf','18','-threads','2','-pix_fmt','yuv420p','-video_track_timescale','60000',str(temp)]
    began=time.monotonic();last=began
    with (cache/f'{index:02d}.log').open('w',encoding='utf-8') as log:
        proc=subprocess.Popen(cmd,stdin=subprocess.PIPE,stdout=subprocess.DEVNULL,stderr=log,creationflags=0x08000000|0x00004000)
        try:
            for k in range(total):
                proc.stdin.write(frame(start+k/FPS).tobytes())
                now=time.monotonic()
                if now-last>25:
                    print(f'{SCENES[index][2]}: {100*(k+1)/total:.0f}% rendered',flush=True);last=now
            proc.stdin.close()
            if proc.wait():raise RuntimeError('Segment '+str(index)+' failed; inspect its log.')
            temp.replace(dest)
        except BaseException:
            if proc.poll() is None:proc.terminate()
            raise
    print(f'Finished {SCENES[index][2]} ({end-start}s)',flush=True)
    return str(dest)

def render_parallel(workers=4):
    if not (ROOT/'original-soundtrack.wav').exists():build_audio()
    files={};began=time.monotonic()
    with ProcessPoolExecutor(max_workers=workers) as pool:
        pending={pool.submit(render_segment,i):i for i in range(len(SCENES))}
        for future in as_completed(pending):
            idx=pending[future];files[idx]=Path(future.result())
            print(f'Completed {len(files)}/{len(SCENES)} chapters · elapsed {time.monotonic()-began:.0f}s',flush=True)
    concat=ROOT/'.render-cache/concat.txt'
    concat.write_text('\n'.join("file '"+files[i].name+"'" for i in range(len(SCENES)))+'\n',encoding='utf-8')
    dest=ROOT/'NetflixPro-TV-3min-1080p60.mp4';temp=dest.with_name(dest.stem+'.rendering.mp4')
    cmd=[str(FFMPEG),'-y','-hide_banner','-loglevel','warning','-f','concat','-safe','0','-i',str(concat),'-i',str(ROOT/'original-soundtrack.wav'),'-i',str(ROOT/'chapters.ffmetadata'),'-map','0:v:0','-map','1:a:0','-map_metadata','2','-map_chapters','2','-c:v','copy','-c:a','aac','-b:a','192k','-ar','48000','-movflags','+faststart','-t','180',str(temp)]
    with (ROOT/'render.log').open('w',encoding='utf-8') as log:
        subprocess.run(cmd,stdout=subprocess.DEVNULL,stderr=log,check=True,creationflags=0x08000000|0x00004000)
    temp.replace(dest)
    print('SAVED '+str(dest),flush=True)
